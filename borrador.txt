@app.route('/horas', methods=['GET'])
def horas():
    selected_day = request.args.get('day')
    if selected_day:
        session['day'] = selected_day
        hoy = datetime.now()
        dias_semana_espanol = ["lunes", "martes", "miércoles", "jueves", "viernes", "sábado", "domingo"]
        if selected_day == 'HOY':
            session['date'] = hoy.strftime('%d/%m/%Y')
            session['weekday'] = dias_semana_espanol[hoy.weekday()]
        elif selected_day == 'MAÑANA':
            manana = hoy + timedelta(days=1)
            session['date'] = manana.strftime('%d/%m/%Y')
            session['weekday'] = dias_semana_espanol[manana.weekday()]
    return render_template('horas.html')

@app.route('/canchas', methods=['GET'])
def canchas():
    selected_time = request.args.get('time')
    if selected_time:
        session['time'] = selected_time
    
    day = session.get('day', 'default_day')  # Asegura que siempre haya un valor por defecto
    time = session.get('time', 'default_time')
    
    # Obtener todas las reservas para el día y la hora seleccionados
    reservas = Reserva.query.filter_by(fecha=session.get('date'), hora=selected_time).all()
    
    # Inicializar el estado de las canchas como disponibles
    canchas_estado = {
        f'cancha{i}': {'estado': 'disponible', 'reservado_por': None} for i in range(1, 7)
    }
    
    # Actualizar el estado con las reservas existentes
    for reserva in reservas:
        canchas_estado[reserva.cancha]['estado'] = 'reservado'
        canchas_estado[reserva.cancha]['reservado_por'] = reserva.reservado_por
    
    return render_template('canchass.html', day=day, time=time, canchas_estado=canchas_estado)

@app.route("/adminn", methods=["GET", "POST"])
def adminn():
    usuario_encontrado = None
    cedula_encontrada = None
    usuario_buscar = None

    # Obtener usuarios desde la base de datos
    usuarios_db = Usuario.query.all()

    # Convertir usuarios en diccionario {nombre: cedula}
    nombres_identificaciones = {usuario.nombre.lower(): [usuario.cedula, usuario.rol] for usuario in usuarios_db}

    if request.method == "POST" and "buscador_nombres" in request.form:
        usuario_buscar = request.form.get("nombre_buscar", "").strip().lower()

        # Buscar ignorando mayúsculas y minúsculas
        if usuario_buscar in nombres_identificaciones:
            usuario_encontrado = usuario_buscar
            cedula_encontrada = nombres_identificaciones[usuario_buscar][0]
        else:
            flash("El usuario no está en la lista.", "error")

    return render_template("buscador_usuario.html", 
                        usuario_encontrado=usuario_encontrado, 
                        cedula_encontrada=cedula_encontrada, 
                        usuario_buscar=usuario_buscar, 
                        usuarios=nombres_identificaciones)

@app.route('/cancelar_reserva/<int:reserva_id>', methods=['POST'])
def cancelar_reserva(reserva_id):
    reserva = Reserva.query.get(reserva_id)
    
    if reserva:
        # Verificar que el usuario solo pueda cancelar antes de la hora reservada
        hora_actual = datetime.now().strftime("%H:%M")
        if hora_actual < reserva.hora:
            db.session.delete(reserva)
            db.session.commit()
            flash("Reserva cancelada con éxito.", "success")
        else:
            flash("No puedes cancelar la reserva después de la hora reservada.", "error")
